# AirPoint — demo commands (run from ~/Projects/airpoint)

# 1. Web dashboard
open index.html

# 2. Numbers for the dashboard (real data + labelled simulation, ~10 s) and the lab
venv/bin/python -m airpoint.results            # writes results.js
venv/bin/python -m airpoint.lab                # tutorial + pointing test at http://127.0.0.1:8765
#    in the lab: enter a code, pick AirPoint / mouse / trackpad, run the test; results/trials.jsonl keeps every block
#    (the midterm benchmark is kept at legacy/midterm_benchmark_suite.py; the dashboard no longer reads it)

# 3. Live demo (q quit | m mouse | d dwell | w wheel | o glow | f face | b filter | k rules/mlp | c calibrate | v mini | z zoom | x x-ray |
#               t simulated tremor, [ ] size, { } speed | p pause | h hints | s sounds | n new sign | r record | 1-6 pose | space start/redo)
venv/bin/python run_demo.py                    # MLP if models/pose_mlp.joblib exists, else rules
venv/bin/python run_demo.py --mouse            # same, OS mouse control already ON
venv/bin/python run_demo.py --rules            # force rules
venv/bin/python run_demo.py --user <name>      # loads profiles/<name>.json (fc_min + personal model) if calibrated
venv/bin/python run_demo.py --record --user <name>   # record poses to data/<name>.csv
#    palm = move | pinch/fist < 0.5 s = click | hold = drag | v-sign 0.3 s = right click
#    thumb up / down = scroll | hand out of frame = rest
#    face demo  : m -> f (look at the screen for a second) -> turn your head: cursor follows; open mouth = click;
#                 raise eyebrows = right click; hold still = wheel; c = calibrate your face; f = back to the hand
#    wheel demo : m -> w -> hold the palm still 1 s: a wheel opens at the cursor; stay = click, or move toward an item
#    teach demo : n -> type a name -> hold a new sign 8 s -> 1 (speak) -> type a phrase -> Enter; hold the sign: it speaks
#    tremor demo: m (mouse on) -> t (cursor shakes, tag reads SIMULATED) -> wait 2 s for LOCKED ->
#                 b twice (smoothing only, then off) to compare on the real cursor -> x to show why

# 3b. Your profile (made by calibration; edit by hand if a sign is hard for you)
venv/bin/python -m airpoint.profile <name>                                   # show
venv/bin/python -m airpoint.profile <name> --role right_click=fist --role grip=pinch --dwell on

# 4. Train on data/*.csv -> models/
venv/bin/python -m airpoint.train
